from aioshad import Client, F
from aioshad.filters import IsMe
from aioshad.types import Message

client = Client(session="my_account")

@client.on_message(IsMe(), F.text == "!ping")
async def ping_handler(msg: Message):
    await msg.edit("Pong! aioshad is active.")

client.run()

اتصال مستقیم به سرورهای لبه شاد

ارتباط بومی با شبکه داخلی شاد بدون مرورگر یا پروکسی میانی.

رمزنگاری معکوس‌شده AES-256

پیاده‌سازی نیتیو رمزگذاری، کلیدسازی RSA و پروتکل امنیتی شاد.

پشتیبانی از چندین اکانت

مدیریت همزمان ده‌ها اکانت و سلف‌بات در یک حلقه رخداد یکپارچه.

ویژگی‌های مهندسی‌شده برای مقیاس‌پذیری

طراحی‌شده از پایه برای سرعت، قابلیت اطمینان و کدهای سازمان‌یافته.

معماری الهام‌گرفته از aiogram ۳

استفاده از روترها (Routers)، دیسپچر، فیلترهای جادویی F و ساختار ماژولار که توسعه بات‌های بزرگ را تمیز و لذت‌بخش می‌کند.

مدیریت سشن‌های منعطف

ذخیره‌سازی آسان کلیدها با فایل سشن (JSON)، پایگاه‌داده امن SQLite و حافظه موقت Memory برای تست و استقرار ابری.

مکالمات خطی و ماشین وضعیت (FSM)

ایجاد پرسش‌نامه‌ها و فرم‌های مرحله‌به‌مرحله با کانتکست منیجر گفت‌وگو و ماشین وضعیت کامل برای سناریوهای چندمرحله‌ای.

پشتیبانی از ویس چت گروهی

امکان ایجاد، پیوستن، مدیریت وضعیت میکروفون و کنترل نشست‌های صوتی WebRTC در گروه‌های شاد به صورت خودکار.

آپلود و دانلود سریع مدیا

انتقال چانک‌شده و استریم فایل‌ها، تصاویر، اسناد و ویس‌ها با محاسبه اتوماتیک تامبنیل و هش‌های رمزنگاری شده.

ابزار خط فرمان (CLI) اختصاصی

راه‌اندازی فوری پروژه‌ها با aioshad init، لاگین مستقیم در ترمینال و مشاهده اطلاعات سشن بدون نیاز به نوشتن کد اولیه.

کدهای تمیز و الگوهای آماده

نمونه‌هایی واقعی از نحوه پیاده‌سازی قابلیت‌ها در aioshad.

from aioshad import Router, F
from aioshad.filters import Command, CommandObject
from aioshad.types import Message

router = Router()

@router.message(Command("calc"))
async def calc_handler(msg: Message, command: CommandObject):
    expression = command.args
    await msg.reply(f"Result: {eval(expression)}")

@router.message(F.text.startswith("info"))
async def info_handler(msg: Message):
    await msg.reply(f"Chat ID: {msg.chat_id}
Author: {msg.author_guid}")
from aioshad import Client, F
from aioshad.types import Message

client = Client(session="bot")

@client.on_message(F.text == "!survey")
async def survey_handler(msg: Message):
    async with client.conversation(chat_id=msg.object_guid, timeout=60) as conv:
        await conv.send("نام شما چیست؟")
        name_msg = await conv.get_response()
        
        await conv.send(f"خوشبختم {name_msg.text}! چند سال دارید؟")
        age_msg = await conv.get_response()
        
        await conv.send(f"ثبت شد: {name_msg.text} ({age_msg.text} سال)")
import asyncio
from aioshad import Client, ClientManager, F
from aioshad.types import Message

manager = ClientManager()

acc1 = Client(session="support_1")
acc2 = Client(session="support_2")

@acc1.on_message(F.text == "ping")
async def h1(msg: Message):
    await msg.reply("Bot 1 online")

manager.add_client(acc1)
manager.add_client(acc2)

if __name__ == "__main__":
    asyncio.run(manager.start_all())
