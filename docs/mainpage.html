فریم‌ورک مدرن و سریع پایتون برای شاد.

کاملاً async، با معماری الهام‌گرفته از aiogram ۳، اتصالات مالتی‌پلکس HTTP/2 و رمزنگاری نیتیو AES-256 و RSA-1024.

Edge 14ms
from aioshad import Client, Router, F
from aioshad.filters import Command
from aioshad.types import Message

client = Client(session="my_bot")
router = Router()

@router.message(Command("start"))
async def on_start(msg: Message):
    await msg.reply("aioshad engine online.")

@router.message(F.text.startswith("/ask"))
async def on_question(msg: Message):
    async with client.conversation(msg.chat_id) as conv:
        await conv.send("پرسش خود را وارد کنید:")
        ans = await conv.get_response()
        await conv.send(f"دریافت شد: {ans.text}")

client.include_router(router)
client.run()

معماری الهام‌گرفته از aiogram ۳

روترهای تودرتو، فیلترهای جادویی F و ساختار کاملاً ماژولار بدون وابستگی‌های پیچیده.

from aioshad import Router, F
from aioshad.types import Message

router = Router()

@router.message(F.text == "!ping")
async def ping_cmd(msg: Message):
    await msg.reply("Pong!")

رمزنگاری معکوس‌شده نیتیو

پیاده‌سازی پرسرعت AES-256-CBC و کلیدسازی RSA-1024 بدون نیاز به مرورگر headless.

from aioshad.crypto import encrypt_payload

# Direct edge payload encryption
enc = encrypt_payload(data, aes_key)

انواع ذخیره‌ساز سشن

پشتیبانی از FileSession، پایگاه‌داده ACID SQLite و MemorySession برای استقرار ابری.

from aioshad.session import SQLiteSessionStorage

storage = SQLiteSessionStorage("bot.db")
client = Client(session=storage)

manager = ClientManager()

manager.add_client(bot_sales)

from aioshad import Router, F
from aioshad.filters import Command, CommandObject
from aioshad.types import Message

router = Router()

@router.message(Command("calc"))
async def calc_handler(msg: Message, command: CommandObject):
    expression = command.args
    await msg.reply(f"Result: {eval(expression)}")

@router.message(F.text.startswith("info"))
async def info_handler(msg: Message):
    await msg.reply(f"Chat ID: {msg.chat_id}
Author: {msg.author_guid}")
from aioshad import Client, F
from aioshad.types import Message

client = Client(session="bot")

@client.on_message(F.text == "!survey")
async def survey_handler(msg: Message):
    async with client.conversation(chat_id=msg.object_guid, timeout=60) as conv:
        await conv.send("نام شما چیست؟")
        name_msg = await conv.get_response()
        
        await conv.send(f"خوشبختم {name_msg.text}! چند سال دارید؟")
        age_msg = await conv.get_response()
        
        await conv.send(f"ثبت شد: {name_msg.text} ({age_msg.text} سال)")
