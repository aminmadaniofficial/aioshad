examples/02_commands_and_routers.py 
import asyncio
from aioshad import Client, Dispatcher, Router
from aioshad.filters import Command, CommandObject, IsGroup
from aioshad.types import Message

user_router = Router(name="user_router")
group_router = Router(name="group_router")

@user_router.message(Command("start", "help", prefix="/!."))
async def handle_start(msg: Message, command: CommandObject):
    await msg.reply(f"Command: {command.command}, Args: {command.args}")

@group_router.message(IsGroup(), Command("info", prefix="!"))
async def handle_group_info(msg: Message):
    chat = await msg.get_chat()
    await msg.reply(f"Group: {chat.title} ({chat.members_count} members)")

async def main():
    dp = Dispatcher()
    dp.include_router(user_router)
    dp.include_router(group_router)
    client = Client(session="my_account", dispatcher=dp)
    await client.start()

if __name__ == "__main__":
    asyncio.run(main())

examples/06_anti_spam_moderator.py 
import re
from aioshad import Client
from aioshad.filters import IsGroup
from aioshad.types import Message

client = Client(session="moderator_account")
LINK_REGEX = re.compile(r"(https?://\S+|shad\.ir/\S+)", re.IGNORECASE)

@client.on_message(IsGroup())
async def anti_link_checker(msg: Message):
    if msg.text and LINK_REGEX.search(msg.text):
        try:
            await msg.delete(delete_type="Global")
        except Exception as exc:
            print(f"Failed to delete: {exc}")

if __name__ == "__main__":
    client.run()

Architecture Comparison

Technical comparison between aioshad and traditional sync/legacy Shad scripts.

Feature aioshad Legacy Scripts
Network Transport HTTP/2 Multiplexing (httpx + h2) HTTP/1.1 blocking (requests)
Cryptography Implementation Native AES-256-CBC + RSA-1024 OAEP Hardcoded / incomplete auth
Gateway Resilience Automatic Multi-Host Failover (*.iranlms.ir) Single hardcoded host
Concurrency Model Pure AsyncIO Event Loop Blocking / Threading pool
Dispatcher & Filtering Aiogram 3 Routers & Magic Filter F Manual if/elif chain
Conversations / FSM async with client.conversation(...) Global dictionaries & flags
WebRTC Voice Chat Full support (create, join, status, users) Unsupported
Multi-Account Support Native ClientManager Manual multiprocessing
